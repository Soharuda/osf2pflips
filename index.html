async function submitQuickSale() { const entryId = safeInt(document.getElementById('quickSaleEntryId').value); const qty = Math.max(1, safeInt(document.getElementById('quickSaleQtyInput').value)); const sellPrice = safeInt(document.getElementById('quickSalePriceInput').value); let purchase = userPurchasesHistory.find(p => p.id === entryId); if (!purchase) { alert("Could not locate purchase batch."); closeQuickSaleModal(); return; } if (qty > purchase.remainingQuantity) { alert(`Cannot sell more than the remaining quantity in this batch (${purchase.remainingQuantity}).`); return; } purchase.remainingQuantity -= qty; let tax = sellPrice > 0 ? Math.min(Math.floor((sellPrice * qty) * 0.01), 5000000) : 0; let netProfit = (sellPrice * qty) - (purchase.buyPrice * qty) - tax; let totalInvestmentCost = purchase.buyPrice * qty; let saleEntry = { id: Date.now(), quantity: qty, sellPrice: sellPrice, buyPrice: purchase.buyPrice, profit: netProfit, totalCost: totalInvestmentCost, date: new Date().toLocaleDateString() + ' ' + new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) }; purchase.salesLog.unshift(saleEntry); userSalesHistory.unshift({ id: saleEntry.id, itemId: purchase.itemId, name: purchase.name, icon: purchase.icon, quantity: qty, sellPrice: sellPrice, buyPrice: purchase.buyPrice, profit: netProfit, totalCost: totalInvestmentCost, date: saleEntry.date }); await saveUserDataToSupabase(); await loadUserDataFromSupabase(); closeQuickSaleModal(); renderGainsView(); await updateHomeHeaderStats(); }
